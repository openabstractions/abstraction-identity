cmake_minimum_required(VERSION 3.16)
project(fileid_installed_module_probe LANGUAGES CXX)
find_package(abstraction_fileid CONFIG REQUIRED)
find_package(abstraction_cas CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(installed_probe_host ../host.cpp)
target_compile_features(installed_probe_host PRIVATE cxx_std_17)
target_link_libraries(installed_probe_host PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(installed_probe_host PROPERTIES ENABLE_EXPORTS ON)
enable_testing()
foreach(kind IN ITEMS fileid cas)
    add_library(${kind}_installed_probe MODULE module.cpp)
    target_compile_features(${kind}_installed_probe PRIVATE cxx_std_17)
    target_link_libraries(${kind}_installed_probe PRIVATE abstraction::fileid)
    if(kind STREQUAL "cas")
        target_compile_definitions(${kind}_installed_probe PRIVATE OA_PROBE_CAS)
        target_link_libraries(${kind}_installed_probe PRIVATE abstraction::cas)
    endif()
    if(APPLE)
        target_link_options(${kind}_installed_probe PRIVATE "-undefined" "dynamic_lookup")
    endif()
    add_test(NAME InstalledModule_${kind} COMMAND installed_probe_host installed $<TARGET_FILE:${kind}_installed_probe> ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(InstalledModule_${kind} PROPERTIES TIMEOUT 12)
endforeach()
