cmake_minimum_required(VERSION 3.16)
project(fileid_installed_consumer LANGUAGES CXX)
find_package(abstraction_fileid 0.8 CONFIG REQUIRED)
if(TARGET abstraction::ipc)
    message(FATAL_ERROR "FileID package must not import the IPC target")
endif()
add_executable(fileid_installed_consumer main.cpp)
target_link_libraries(fileid_installed_consumer PRIVATE abstraction::fileid)
enable_testing()
add_test(NAME InstalledFileID COMMAND fileid_installed_consumer)
