cmake_minimum_required(VERSION 3.16)
project(ipc_installed_consumer LANGUAGES C CXX)
find_package(abstraction_ipc CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(ipc_c_consumer main.c fixture.cpp)
target_compile_features(ipc_c_consumer PRIVATE cxx_std_17)
target_link_libraries(ipc_c_consumer PRIVATE abstraction::ipc Threads::Threads)
enable_testing()
add_test(NAME CConsumer COMMAND ipc_c_consumer)
set_tests_properties(CConsumer PROPERTIES TIMEOUT 10)

option(TEST_DOWNLOAD_PACKAGE "Also consume the installed discovery target" OFF)
if(TEST_DOWNLOAD_PACKAGE)
    find_package(abstraction_download CONFIG REQUIRED)
    add_executable(ipc_discovery_consumer discovery.cpp)
    target_link_libraries(ipc_discovery_consumer PRIVATE abstraction::discovery)
    add_test(NAME InstalledDiscovery COMMAND ipc_discovery_consumer)
    set_tests_properties(InstalledDiscovery PROPERTIES TIMEOUT 10)
endif()
