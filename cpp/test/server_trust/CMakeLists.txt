cmake_minimum_required(VERSION 3.16)
project(ipc_installed_server_trust LANGUAGES C CXX)
find_package(abstraction_ipc CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(ipc_server_trust main.c fixture.cpp)
target_compile_features(ipc_server_trust PRIVATE cxx_std_17)
target_link_libraries(ipc_server_trust PRIVATE abstraction::ipc Threads::Threads)
enable_testing()
add_test(NAME InstalledServerTrust COMMAND ipc_server_trust)
set_tests_properties(InstalledServerTrust PROPERTIES TIMEOUT 20)

add_executable(ipc_wrapper_trust wrapper.cpp fixture.cpp)
target_compile_features(ipc_wrapper_trust PRIVATE cxx_std_17)
target_link_libraries(ipc_wrapper_trust PRIVATE abstraction::ipc Threads::Threads)
add_test(NAME InstalledWrapperTrust COMMAND ipc_wrapper_trust)
set_tests_properties(InstalledWrapperTrust PROPERTIES TIMEOUT 20)
