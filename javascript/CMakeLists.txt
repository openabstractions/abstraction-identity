cmake_minimum_required(VERSION 3.16)
project(abstraction_ipc_node LANGUAGES CXX)
find_package(abstraction_ipc CONFIG REQUIRED)
get_target_property(ipc_type abstraction::ipc TYPE)
if(NOT ipc_type STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "Node addon requires a static abstraction_ipc package")
endif()
set(NODE_INCLUDE_DIR "" CACHE PATH "Directory containing official node_api.h")
if(NOT EXISTS "${NODE_INCLUDE_DIR}/node_api.h")
  message(FATAL_ERROR "NODE_INCLUDE_DIR must contain Node-API headers")
endif()
add_library(oa_ipc_node MODULE addon.cpp)
target_include_directories(oa_ipc_node PRIVATE "${NODE_INCLUDE_DIR}")
target_compile_features(oa_ipc_node PRIVATE cxx_std_17)
target_compile_definitions(oa_ipc_node PRIVATE NAPI_VERSION=8 NODE_GYP_MODULE_NAME=oa_ipc_node)
target_link_libraries(oa_ipc_node PRIVATE abstraction::ipc)
if(WIN32)
  set(NODE_IMPORT_LIBRARY "" CACHE FILEPATH "Official matching architecture node.lib")
  if(NOT EXISTS "${NODE_IMPORT_LIBRARY}")
    message(FATAL_ERROR "NODE_IMPORT_LIBRARY must name node.lib")
  endif()
  target_link_libraries(oa_ipc_node PRIVATE "${NODE_IMPORT_LIBRARY}")
elseif(APPLE)
  target_link_options(oa_ipc_node PRIVATE "-undefined" "dynamic_lookup")
endif()
set_target_properties(oa_ipc_node PROPERTIES PREFIX "" SUFFIX ".node")
install(TARGETS oa_ipc_node LIBRARY DESTINATION native RUNTIME DESTINATION native)
install(FILES package.json index.js README.md DESTINATION .)
